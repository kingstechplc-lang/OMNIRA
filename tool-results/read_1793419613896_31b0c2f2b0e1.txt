     3→// Copyright KINGSTECH PLC. Evolved from the PHARMORA platform
     4→// architecture ("One Platform. Every Pharmacy." → "One Platform.
     5→// Every Business.").
     6→//
     7→// Conventions (inherited from Pharmora)
     8→//  - UUID primary keys (client-generated via Prisma uuid())
     9→//  - Every tenant-owned entity carries organizationId (tenant isolation)
    10→//  - Composite unique constraints scoped per tenant (never global)
    11→//  - Money as Decimal (14,2-equivalent), tax rates as Decimal — never
    12→//    floating point in the database
    13→//  - SQLite: enum-like fields are String constants validated by Zod in
    14→//    the API layer (see src/lib/constants.ts)
    15→//
    16→// Universal business support (new in OMNIRA)
    17→//  - Organization.businessType — a retail vertical pack (supermarket,
    18→//    pharmacy, restaurant, fashion, electronics, hardware, beauty,
    19→//    liquor, bookstore, wholesale, service, general)
    20→//  - Product.trackBatches — perishable/expiry tracking is optional per
    21→//    product (on by default for pharmacy/grocery packs, off elsewhere)
    22→//  - Loyalty accrual on customers; tax per product; multi-warehouse
    23→// =====================================================================

   595→model Customer {
   596→  id             String   @id @default(uuid())
   597→  organizationId String
   598→  name           String
   599→  phone          String?
   600→  email          String?
   601→  addressLine1   String?

   606→  visitCount     Int      @default(0)
   607→  isActive       Boolean  @default(true)
   608→  createdAt      DateTime @default(now())
   609→  updatedAt      DateTime @updatedAt
   610→
   611→  organization Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)

   618→model Sale {
   619→  id              String   @id @default(uuid())
   620→  organizationId  String
   621→  branchId        String?
   622→  warehouseId     String?
   623→  saleNumber      String
   624→  cashierId       String?
   625→  customerId      String?
   626→  clientRequestId String? // POS idempotency key (replay safety)
   627→  source          String   @default("POS") // POS | ONLINE | IMPORT
   628→
   629→  subtotal        Decimal  @default(0)
   630→  taxTotal        Decimal  @default(0)
   631→  discountTotal   Decimal  @default(0)
   632→  grandTotal      Decimal  @default(0)
   633→  costTotal       Decimal  @default(0) // COGS snapshot for profit reporting
   634→
   635→  status          String   @default("COMPLETED") // COMPLETED | VOIDED | REFUNDED | PARTIALLY_REFUNDED
   636→  voidedById      String?
   637→  voidedAt        DateTime?
   638→  voidReason      String?
   639→  createdAt       DateTime @default(now())
   640→  updatedAt       DateTime @updatedAt
   641→
   642→  organization Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
   643→  branch       Branch?      @relation(fields: [branchId], references: [id], onDelete: SetNull)
   644→  cashier      User?        @relation("SaleCashier", fields: [cashierId], references: [id], onDelete: SetNull)
   645→  customer     Customer?    @relation(fields: [customerId], references: [id], onDelete: SetNull)
   646→  voidedBy     User?        @relation("SaleVoider", fields: [voidedById], references: [id], onDelete: SetNull)
   647→  items        SaleItem[]
   648→  payments     Payment[]
   649→  refunds      Refund[]
   650→
   651→  @@unique([organizationId, saleNumber])
   652→  @@unique([organizationId, clientRequestId])
   653→  @@index([organizationId, createdAt])
   654→  @@index([branchId, createdAt])
   655→  @@index([customerId])
   656→  @@index([status])
   657→}

   659→model SaleItem {

   670→  lineDiscount Decimal @default(0)
   671→  taxRate     Decimal  @default(0)
   672→  lineTotal   Decimal  @default(0)
   673→
   674→  sale    Sale     @relation(fields: [saleId], references: [id], onDelete: Cascade)
   675→  product Product? @relation(fields: [productId], references: [id], onDelete: SetNull)
   676→  refundItems RefundItem[]
   677→
   678→  @@index([saleId])
   679→  @@index([productId])
   680→}

   694→  @@index([saleId])
