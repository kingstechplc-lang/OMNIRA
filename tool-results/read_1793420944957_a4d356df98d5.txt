    98→export function useAuthContext() {
    99→  return useQuery<AuthContextDTO>({
   100→    queryKey: queryKeys.authContext,

   103→    retry: false,
   104→  });

   107→export function useSwitchOrganization() {
   108→  const qc = useQueryClient();
   109→  return useMutation({
   110→    mutationFn: (organizationId: string | null) =>
   111→      post<AuthContextDTO>('/api/auth/switch-organization', { organizationId }),
   112→    onSuccess: () => {
   113→      // Full refetch of every tenant-scoped query.
   114→      qc.clear();
   115→      void qc.refetchQueries({ queryKey: queryKeys.authContext });
   116→    },
   117→  });
   118→}
