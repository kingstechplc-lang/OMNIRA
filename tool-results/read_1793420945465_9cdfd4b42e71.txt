   107→   107→export function useSwitchOrganization() {
   108→   108→  const qc = useQueryClient();
   109→   109→  return useMutation({
   110→   110→    mutationFn: (organizationId: string | null) =>
   111→   111→      post<AuthContextDTO>('/api/auth/switch-organization', { organizationId }),
   112→   112→    onSuccess: () => {
   113→   113→      // Full refetch of every tenant-scoped query.
   114→   114→      qc.clear();
   115→   115→      void qc.refetchQueries({ queryKey: queryKeys.authContext });
   116→   116→    },
   117→   117→  });
   118→   118→}
